cmake_minimum_required(VERSION 3.20)
project(hisense_native_tests LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
if(MSVC)
  add_compile_options(/EHsc)
endif()
set(COMPONENT "${CMAKE_CURRENT_SOURCE_DIR}/../components/hisense_ac")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
             "${COMPONENT}/hisense_ac.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/generate_native.py")
option(ENABLE_SANITIZERS "Run host tests with address/undefined sanitizers" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(ENABLE_SANITIZERS)
  message(FATAL_ERROR "Sanitizers require a supported GCC/Clang host toolchain")
endif()
execute_process(
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_native.py"
          "${CMAKE_CURRENT_BINARY_DIR}"
  COMMAND_ERROR_IS_FATAL ANY)
add_executable(command_tests "${CMAKE_CURRENT_BINARY_DIR}/command_tests.cpp"
               "${COMPONENT}/commands.cpp")
target_include_directories(command_tests PRIVATE "${COMPONENT}")
add_test(NAME command_golden COMMAND command_tests)
foreach(test IN ITEMS protocol_tests protocol_unsigned_tests response_adapter_tests transport_tests state_sync_tests)
  if(test STREQUAL "protocol_unsigned_tests")
    set(source protocol_tests)
  else()
    set(source "${test}")
  endif()
  add_executable("${test}" "${source}.cpp" "${COMPONENT}/protocol.cpp")
  target_include_directories("${test}" PRIVATE "${COMPONENT}" "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions("${test}" PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
  add_test(NAME "${test}" COMMAND "${test}")
endforeach()
target_sources(transport_tests PRIVATE "${COMPONENT}/transport.cpp" "${COMPONENT}/commands.cpp")
target_sources(response_adapter_tests PRIVATE "${COMPONENT}/hisense_ac.cpp" "${COMPONENT}/transport.cpp" "${COMPONENT}/commands.cpp")
target_include_directories(response_adapter_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/stubs")
target_sources(state_sync_tests PRIVATE "${COMPONENT}/hisense_ac.cpp" "${COMPONENT}/transport.cpp" "${COMPONENT}/commands.cpp")
target_include_directories(state_sync_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/stubs")
if(MSVC)
  target_compile_options(protocol_unsigned_tests PRIVATE /J)
else()
  target_compile_options(protocol_unsigned_tests PRIVATE -funsigned-char)
  target_compile_options(protocol_tests PRIVATE -fsigned-char)
endif()
